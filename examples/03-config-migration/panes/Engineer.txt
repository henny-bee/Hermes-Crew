+import configparser
+import json
+import os
+import sys
+
+
+def parse_ini_config(ini_path="config.ini"):
+    cp = configparser.ConfigParser()
+    if not cp.read(ini_path):
+        # Even if file is missing/empty, fallbacks should be respected if section/key missing
+        pass
+    return {
+        "host": cp.get("server", "host", fallback="127.0.0.1"),
+        "port": cp.getint("server", "port", fallback=8000),
+        "db_url": cp.get("database", "url", fallback="sqlite:///app.db"),
+        "debug": cp.getboolean("server", "debug", fallback=False),
+    }
+
+
+def migrate(ini_path="config.ini", json_path="config.json"):
+    if os.path.exists(json_path):
+        raise FileExistsError(f"Target file already exists: {json_path}")
+    data = parse_ini_config(ini_path)
+    with open(json_path, "w", encoding="utf-8") as f:
+        json.dump(data, f, indent=2)
+        f.write("\n")
+    return data
+
+
+def main():
+    ini_path = sys.argv[1] if len(sys.argv) > 1 else "config.ini"
+    json_path = sys.argv[2] if len(sys.argv) > 2 else "config.json"
+    try:
+        data = migrate(ini_path, json_path)
+        print(f"Migrated configuration from {ini_path} to {json_path}: {data}")
+    except FileExistsError as e:
+        print(f"Error: {e}", file=sys.stderr)
+        sys.exit(1)
+
+
+if __name__ == "__main__":
+    main()
